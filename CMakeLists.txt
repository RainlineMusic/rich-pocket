cmake_minimum_required(VERSION 3.22)
project(RichPocket VERSION 0.1.0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED JUCE_DIR)
 set(JUCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../JUCE")
endif()
set(DUCK_SANITIZER "none" CACHE STRING "Debug instrumentation: none, address, undefined, thread")
set_property(CACHE DUCK_SANITIZER PROPERTY STRINGS none address undefined thread)
if(NOT DUCK_SANITIZER STREQUAL "none")
 if(MSVC)
  if(NOT DUCK_SANITIZER STREQUAL "address")
   message(FATAL_ERROR "MSVC supports the address configuration here")
  endif()
  add_compile_options(/fsanitize=address)
 else()
  if(DUCK_SANITIZER STREQUAL "address")
   set(DUCK_SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(DUCK_SANITIZER STREQUAL "undefined")
   set(DUCK_SANITIZER_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
  elseif(DUCK_SANITIZER STREQUAL "thread")
   set(DUCK_SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  else()
   message(FATAL_ERROR "Unknown DUCK_SANITIZER")
  endif()
  add_compile_options(${DUCK_SANITIZER_FLAGS})
  add_link_options(${DUCK_SANITIZER_FLAGS})
 endif()
endif()
if(WIN32)
 include(cmake/ApplyWindowsVBlankFix.cmake)
 duck_apply_windows_vblank_fix("${JUCE_DIR}")
endif()
add_subdirectory(${JUCE_DIR} JUCE)
set(DUCK_DEFAULT_FORMATS VST3 AAX Standalone)
if(APPLE)
 list(APPEND DUCK_DEFAULT_FORMATS AU)
endif()
set(DUCK_FORMATS "${DUCK_DEFAULT_FORMATS}" CACHE STRING "Plugin formats")
option(DUCK_ENABLE_CLAP "Build CLAP using clap-juce-extensions" OFF)
if(DUCK_ENABLE_CLAP)
 include(FetchContent)
 FetchContent_Declare(clap_juce_extensions
  GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git
  GIT_TAG 7adee3a1bd4684d4caa5601100e364abccff4b4d
  GIT_SUBMODULES_RECURSE TRUE)
 FetchContent_MakeAvailable(clap_juce_extensions)
endif()
juce_add_plugin(RichPocket COMPANY_NAME "Rainline Music" COMPANY_WEBSITE "https://rainlinemusic.su" COMPANY_EMAIL "rainlinemusic@gmail.com" BUNDLE_ID "ru.rainlinemusic.richpocket" AAX_IDENTIFIER "ru.rainlinemusic.richpocket" PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE RcPk FORMATS ${DUCK_FORMATS} PRODUCT_NAME "Rich Pocket" DESCRIPTION "Four-colour mastering saturation" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Distortion AAX_CATEGORY Harmonic)
set(DUCK_LICENSE_API_URL "" CACHE STRING "HTTPS endpoint for device-bound online activation")
if(DUCK_LICENSE_API_URL AND NOT DUCK_LICENSE_API_URL MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[A-Za-z0-9/_-]*)?$")
 message(FATAL_ERROR "DUCK_LICENSE_API_URL must be an HTTPS URL without query or fragment")
endif()
configure_file(Source/LicenseConfig.h.in generated/LicenseConfig.h @ONLY)
target_include_directories(RichPocket PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
juce_generate_juce_header(RichPocket)
target_sources(RichPocket PRIVATE Source/RichDSP.h Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h)
target_compile_definitions(RichPocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(RichPocket PRIVATE /utf-8)
else()
 target_compile_options(RichPocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
target_link_libraries(RichPocket PRIVATE juce::juce_cryptography juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
option(RICH_ENABLE_LICENSE "Require activation (disabled for development)" OFF)
target_compile_definitions(RichPocket PUBLIC RICH_ENABLE_LICENSE=$<BOOL:${RICH_ENABLE_LICENSE}>)
option(DUCK_ENABLE_OPENGL "Enable inherited OpenGL editor renderer" ON)
if(DUCK_ENABLE_OPENGL)
 target_link_libraries(RichPocket PRIVATE juce::juce_opengl)
endif()
target_compile_definitions(RichPocket PUBLIC DUCK_ENABLE_OPENGL=$<BOOL:${DUCK_ENABLE_OPENGL}>)
enable_testing()
add_executable(RichDSPTest Tests/rich_dsp_test.cpp)
add_test(NAME RichDSP COMMAND RichDSPTest)
option(DUCK_BUILD_UI_TESTS "Build Rich integration and UI capture tests" OFF)
juce_add_binary_data(DuckAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf Assets/Designer/DialLarge.png Assets/Designer/DialSmall.png Assets/Designer/TitleInfluence.svg Assets/Designer/TitleDuration.svg Assets/Designer/TitleOutput.svg Assets/Designer/TitleMS.svg Assets/Designer/Logo.svg Assets/Audio/Duck44100.wav Assets/Audio/Duck48000.wav Assets/Audio/Duck88200.wav Assets/Audio/Duck96000.wav Assets/Audio/Duck176400.wav Assets/Audio/Duck192000.wav)
set_target_properties(DuckAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(RichPocket PUBLIC DuckAssets)
if(DUCK_BUILD_UI_TESTS)
 juce_add_console_app(RichIntegrationTest PRODUCT_NAME "Rich Pocket integration tests")
 juce_generate_juce_header(RichIntegrationTest)
 target_sources(RichIntegrationTest PRIVATE Tests/rich_integration_test.cpp)
 target_include_directories(RichIntegrationTest PRIVATE Source ${CMAKE_CURRENT_BINARY_DIR}/RichPocket_artefacts/JuceLibraryCode)
 target_compile_definitions(RichIntegrationTest PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(RichIntegrationTest PRIVATE RichPocket juce::juce_audio_utils juce::juce_cryptography juce::juce_dsp)
 if(DUCK_ENABLE_OPENGL)
  target_link_libraries(RichIntegrationTest PRIVATE juce::juce_opengl)
 endif()
 add_test(NAME RichIntegration COMMAND RichIntegrationTest "${CMAKE_CURRENT_BINARY_DIR}/ui-captures")
endif()
# Carry the embedded-font license in every delivered plug-in bundle, before
# CI signs/notarises it. Fonts remain embedded; only the license is copied.
foreach(format VST3 AAX AU)
 if(TARGET RichPocket_${format})
  add_custom_command(TARGET RichPocket_${format} POST_BUILD
   COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RichPocket_${format}>/../Resources"
   COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:RichPocket_${format}>/../Resources/Inter-OFL.txt"
   VERBATIM)
 endif()
endforeach()

endif()

  set(DUCK_CLAP_LICENSE_DIR "$<TARGET_FILE_DIR:RichPocket_CLAP>/../Resources/Licenses")

  set(DUCK_CLAP_LICENSE_DIR "$<TARGET_FILE_DIR:RichPocket_CLAP>/Licenses")
 endif()
 add_custom_command(TARGET RichPocket_CLAP POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${DUCK_CLAP_LICENSE_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Licenses" "${DUCK_CLAP_LICENSE_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "${DUCK_CLAP_LICENSE_DIR}/Inter-OFL.txt"
  VERBATIM)
endif()
